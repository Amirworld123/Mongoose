// 1. Create an explicit "books" collection with validation: title must exist, be a string, and non-empty
db.createCollection('books', {
    validator: {
        title: { $exists: true, $type: 'string', $ne: '' }
    }
})

// 2. Create an implicit "authors" collection by inserting directly
db.authors.insertOne({ name: 'Author1', nationality: 'British' })

// 3. Create a capped "logs" collection with a 1MB size limit
db.createCollection('logs', { capped: true, size: 1048576 })

// 4. Create an index on the "title" field
db.books.createIndex({ title: 1 })

// 5. Insert one document into the "books" collection
db.books.insertOne({
    title: 'Book1',
    author: 'Ali',
    year: 1937,
    genre: ['Fantasy', 'Adventure']
})

// 6. Insert more books, including "Future" and "Brave New World" needed by later queries
db.books.insertMany([
    {
        title: 'Book2',
        author: 'Ahmed',
        year: 2001,
        genre: ['Drama']
    },
    {
        title: 'Book3',
        author: 'Omar',
        year: 2010,
        genre: ['Mystery', 'Thriller']
    },
    {
        title: 'Book4',
        author: 'Sara',
        year: 2020,
        genre: ['Romance']
    },
    {
        title: 'Brave New World',
        author: 'Aldous Huxley',
        year: 1932,
        genre: ['Science Fiction', 'Dystopian']
    },
    {
        title: 'Future',
        author: 'Ali',
        year: 2021,
        genre: ['Science Fiction']
    }
])

// 7. Insert a log entry, referencing a real book's ObjectId (not a plain string)
//    Using "Book2" (year 2001) since it survives the deleteMany step further down
const loggedBook = db.books.findOne({ title: 'Book2' })
db.logs.insertOne({ book_id: loggedBook._id, action: 'borrowed' })

// 8. Update the book with title "Future", change its year to 2022
db.books.updateOne({ title: 'Future' }, { $set: { year: 2022 } })

// 9. Find the book with title "Brave New World"
db.books.find({ title: 'Brave New World' })

// 10. Find all books published between 1990 and 2010
db.books.find({ year: { $gte: 1990, $lte: 2010 } })

// 11. Find books with genre "Science Fiction"
db.books.find({ genre: { $in: ['Science Fiction'] } })

// 12. Skip the first 2, limit to 3, sorted by year ascending
db.books.find().skip(2).limit(3).sort({ year: 1 })

// 13. Find books where year is stored as an integer
db.books.find({ year: { $type: 'int' } })

// 14. Find books that do NOT contain "Horror" or "Science Fiction"
db.books.find({ genre: { $nin: ['Horror', 'Science Fiction'] } })

// 15. Delete all books published before 2000
db.books.deleteMany({ year: { $lt: 2000 } })

// 16. Aggregation: filter books published after 2000, sort by year descending
db.books.aggregate([
    { $match: { year: { $gt: 2000 } } },
    { $sort: { year: -1 } }
])

// 17. Aggregation: filter books after 2000, show only title, author, year
db.books.aggregate([
    { $match: { year: { $gt: 2000 } } },
    {
        $project: {
            _id: 0,
            title: 1,
            author: 1,
            year: 1
        }
    }
])

// 18. Aggregation: break the genre array into separate documents
db.books.aggregate([
    {
        $unwind: {
            path: '$genre',
            includeArrayIndex: 'TagIndex',
            preserveNullAndEmptyArrays: true
        }
    }
])

// 19. Aggregation: join books with logs on _id / book_id
db.books.aggregate([
    {
        $lookup: {
            from: 'logs',
            localField: '_id',
            foreignField: 'book_id',
            as: 'logs'
        }
    }
])
